## Why Use Rust for AI Backends?

AI applications are typically bottlenecked by network I/O (calling LLM APIs) rather than CPU — which is precisely where Rust's async model shines:

- **Handle massive concurrent AI requests**: Tokio can process thousands of LLM requests simultaneously with minimal memory usage
- **Streaming responses**: Rust's `Stream` trait is a natural fit for SSE (Server-Sent Events)
- **Vector computation**: Rust's SIMD support makes vector similarity calculations 10x+ faster than Python
- **Edge deployment**: Rust binaries can be deployed to Cloudflare Workers and Fly.io edge nodes

## Streaming Responses: SSE + Axum

One of the most important UX features in AI applications — streaming output lets users see tokens generated word by word:

```toml
[dependencies]
axum          = "0.8"
reqwest       = { version = "0.12", features = ["stream"] }
tokio-stream  = "0.1"
futures       = "0.3"
eventsource-stream = "0.2"
```

```rust
// src/routes/chat.rs — streaming AI response
use axum::{
    extract::Json,
    response::{IntoResponse, Sse},
    response::sse::{Event, KeepAlive},
};
use futures::StreamExt;
use reqwest::Client;
use serde::{Deserialize, Serialize};
use tokio_stream::wrappers::ReceiverStream;
use tokio::sync::mpsc;

#[derive(Deserialize)]
pub struct ChatInput { message: String }

pub async fn stream_chat(
    Json(input): Json<ChatInput>,
) -> impl IntoResponse {
    let (tx, rx) = mpsc::channel::<Result<Event, std::convert::Infallible>>(32);

    // call LLM in a background task and stream the results
    tokio::spawn(async move {
        let client = Client::new();

        let mut stream = client
            .post("https://api.openai.com/v1/chat/completions")
            .bearer_auth(std::env::var("OPENAI_API_KEY").unwrap())
            .json(&serde_json::json!({
                "model": "gpt-4o",
                "stream": true,
                "messages": [{ "role": "user", "content": input.message }]
            }))
            .send().await.unwrap()
            .bytes_stream();

        while let Some(chunk) = stream.next().await {
            if let Ok(bytes) = chunk {
                let text = String::from_utf8_lossy(&bytes);
                // parse SSE data lines and extract delta content
                for line in text.lines() {
                    if let Some(data) = line.strip_prefix("data: ") {
                        if data == "[DONE]" { break; }
                        if let Ok(val) = serde_json::from_str::<serde_json::Value>(data) {
                            if let Some(content) = val
                                .pointer("/choices/0/delta/content")
                                .and_then(|v| v.as_str())
                            {
                                let _ = tx.send(Ok(Event::default().data(content))).await;
                            }
                        }
                    }
                }
            }
        }
    });

    // convert the channel into an SSE stream
    let stream = ReceiverStream::new(rx);
    Sse::new(stream).keep_alive(KeepAlive::default())
}
```

### Consuming SSE on the Frontend

```typescript
// frontend/src/components/ChatStream.tsx
"use client";

import { useState } from "react";

export function ChatStream() {
  const [response, setResponse] = useState("");
  const [loading, setLoading]   = useState(false);

  async function sendMessage(message: string) {
    setResponse("");
    setLoading(true);

    const eventSource = new EventSource(
      `/api/chat/stream?message=${encodeURIComponent(message)}`
    );

    eventSource.onmessage = (e) => {
      setResponse((prev) => prev + e.data);
    };

    eventSource.onerror = () => {
      setLoading(false);
      eventSource.close();
    };
  }

  return (
    <div>
      <button onClick={() => sendMessage("Explain Rust ownership")}>
        Ask AI
      </button>
      <pre className="whitespace-pre-wrap">{response}</pre>
    </div>
  );
}
```

## Vector Database and RAG

Retrieval-Augmented Generation (RAG) — enabling AI to answer questions specific to your codebase:

```toml
[dependencies]
pgvector   = { version = "0.4", features = ["sqlx"] }  # PostgreSQL vector extension
async-openai = "0.28"  # official OpenAI Rust client
```

```sql
-- enable pgvector extension
CREATE EXTENSION IF NOT EXISTS vector;

-- document vector table
CREATE TABLE documents (
    id        SERIAL PRIMARY KEY,
    content   TEXT    NOT NULL,
    metadata  JSONB,
    embedding vector(1536)  -- OpenAI text-embedding-3-small dimensions
);

-- vector index (speeds up similarity search)
CREATE INDEX ON documents USING ivfflat (embedding vector_cosine_ops)
    WITH (lists = 100);
```

```rust
// src/rag.rs — RAG pipeline
use async_openai::{Client, types::{CreateEmbeddingRequestArgs, CreateChatCompletionRequestArgs}};
use pgvector::Vector;
use sqlx::PgPool;

pub struct RagPipeline {
    openai: Client<async_openai::config::OpenAIConfig>,
    pool:   PgPool,
}

impl RagPipeline {
    // 1. embed a document and store it
    pub async fn index_document(&self, content: &str, metadata: serde_json::Value) -> anyhow::Result<()> {
        let embedding = self.embed(content).await?;

        sqlx::query!(
            "INSERT INTO documents (content, metadata, embedding) VALUES ($1, $2, $3)",
            content,
            metadata,
            embedding as Vector,
        )
        .execute(&self.pool)
        .await?;

        Ok(())
    }

    // 2. similarity search
    pub async fn search(&self, query: &str, limit: i64) -> anyhow::Result<Vec<String>> {
        let query_embedding = self.embed(query).await?;

        // pgvector cosine similarity search (<=> operator)
        let rows = sqlx::query!(
            r#"
            SELECT content
            FROM documents
            ORDER BY embedding <=> $1
            LIMIT $2
            "#,
            query_embedding as Vector,
            limit,
        )
        .fetch_all(&self.pool)
        .await?;

        Ok(rows.into_iter().map(|r| r.content).collect())
    }

    // 3. RAG: retrieve + generate
    pub async fn ask(&self, question: &str) -> anyhow::Result<String> {
        // retrieve relevant documents
        let contexts = self.search(question, 5).await?;
        let context = contexts.join("\n\n---\n\n");

        // build a prompt with context
        let prompt = format!(
            "Answer the question based on the following code documentation.\n\nDocumentation:\n{context}\n\nQuestion: {question}"
        );

        // call LLM to generate the answer
        let request = CreateChatCompletionRequestArgs::default()
            .model("gpt-4o")
            .messages([async_openai::types::ChatCompletionRequestUserMessageArgs::default()
                .content(prompt)
                .build()?
                .into()])
            .build()?;

        let response = self.openai.chat().create(request).await?;
        Ok(response.choices[0].message.content.clone().unwrap_or_default())
    }

    async fn embed(&self, text: &str) -> anyhow::Result<Vector> {
        let request = CreateEmbeddingRequestArgs::default()
            .model("text-embedding-3-small")
            .input(text)
            .build()?;

        let response = self.openai.embeddings().create(request).await?;
        let values: Vec<f32> = response.data[0].embedding.clone();
        Ok(Vector::from(values))
    }
}
```

## AI Agent: Tool Calling (Function Calling)

Let the AI autonomously decide which function to call:

```rust
// src/agent.rs — simple AI Agent
use serde_json::{json, Value};

// define tools
const TOOLS: &str = r#"[
  {
    "type": "function",
    "function": {
      "name": "run_cargo_check",
      "description": "Run cargo check on Rust code and return compilation errors",
      "parameters": {
        "type": "object",
        "properties": {
          "code": { "type": "string", "description": "The Rust code to check" }
        },
        "required": ["code"]
      }
    }
  },
  {
    "type": "function",
    "function": {
      "name": "search_crates_io",
      "description": "Search for a Rust crate on crates.io",
      "parameters": {
        "type": "object",
        "properties": {
          "query": { "type": "string" }
        },
        "required": ["query"]
      }
    }
  }
]"#;

pub async fn run_agent(user_request: &str) -> anyhow::Result<String> {
    let client = reqwest::Client::new();
    let tools: Value = serde_json::from_str(TOOLS)?;
    let mut messages = vec![json!({ "role": "user", "content": user_request })];

    loop {
        let response = client
            .post("https://api.openai.com/v1/chat/completions")
            .bearer_auth(std::env::var("OPENAI_API_KEY")?)
            .json(&json!({
                "model": "gpt-4o",
                "messages": messages,
                "tools": tools,
            }))
            .send().await?
            .json::<Value>().await?;

        let choice = &response["choices"][0];
        let message = &choice["message"];

        // no tool call → done
        if choice["finish_reason"] == "stop" {
            return Ok(message["content"].as_str().unwrap_or_default().to_string());
        }

        // tool call present → execute and return results
        if let Some(tool_calls) = message["tool_calls"].as_array() {
            messages.push(message.clone());

            for call in tool_calls {
                let fn_name = call["function"]["name"].as_str().unwrap();
                let args: Value = serde_json::from_str(
                    call["function"]["arguments"].as_str().unwrap()
                )?;

                let result = match fn_name {
                    "run_cargo_check" => execute_cargo_check(&args).await,
                    "search_crates_io" => search_crates(&args).await,
                    _ => "unknown tool".to_string(),
                };

                messages.push(json!({
                    "role": "tool",
                    "tool_call_id": call["id"],
                    "content": result,
                }));
            }
        }
    }
}
```
